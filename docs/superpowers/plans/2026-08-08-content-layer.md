# Content Layer + Minneapolis Equivalence Gate — Implementation Plan (Plan 1 of 3)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move every dynamic content slot (per `docs/superpowers/specs/2026-08-08-dynamic-static-content-list.md`) out of hardcoded strings into a per-city content layer, with proof that the rendered Minneapolis site is byte-equivalent before and after.

**Architecture:** A `CityContent` JSON document (seeded for Minneapolis from today's data) + a token-interpolation helper. Data files in `src/data/` keep their exported names and shapes — only their internals change to derive values from the city document. Components and pages are NOT touched in this plan (their hardcoded "Minneapolis" strings are Plan 2, when the city context threads through). Two gates enforce zero visible change: vitest snapshots of every data module's exported values, and an HTML crawler diff of every route.

**Tech Stack:** Next.js 16.2.12 (existing), TypeScript, vitest (new dev dep), Node 24 (native TS import for scripts), pnpm.

**PROJECT RULE — NO COMMITS.** The user has forbidden git commits without explicit approval. Every task ends at a **Checkpoint** instead of a commit. Do not run `git commit` or `git add`. The user triggers commits.

```js
// scripts/snapshot-pages.mjs
// Usage: node scripts/snapshot-pages.mjs baseline|compare
// Builds nothing itself — expects the prod server already running on PORT.
// Crawls same-origin links from "/", saves normalized HTML per route,
// and in compare mode diffs against the baseline dir.
import fs from 'node:fs'
import path from 'node:path'

const MODE = process.argv[2]
if (!['baseline', 'compare'].includes(MODE)) {
  console.error('usage: node scripts/snapshot-pages.mjs baseline|compare')
  process.exit(2)
}
const BASE = process.env.SNAPSHOT_ORIGIN ?? 'http://localhost:3100'
const OUT = path.join('scripts', '.snapshots', MODE === 'baseline' ? 'baseline' : 'current')
const BASELINE = path.join('scripts', '.snapshots', 'baseline')

function normalize(html) {
  return html
    .replace(/<script\b[^>]*>[\s\S]*?<\/script>/g, '')          // build-dependent payloads
    .replace(/\/_next\/static\/[^"'\s)]+/g, '/_next/static/X')  // hashed asset paths
    .replace(/\snonce="[^"]*"/g, '')
    .replace(/\s+/g, ' ')
    .trim()
}

function routeToFile(route) {
  return (route === '/' ? 'index' : route.replace(/^\//, '').replace(/\//g, '__')) + '.html'
}

const seen = new Set()
const queue = ['/']
const results = new Map()

while (queue.length > 0) {
  const route = queue.shift()
  if (seen.has(route)) continue
  seen.add(route)
  const res = await fetch(BASE + route, { redirect: 'manual' })
  if (res.status !== 200) {
    console.log(`skip ${route} (${res.status})`)
    continue
  }
  const html = await res.text()
  results.set(route, normalize(html))
  for (const m of html.matchAll(/href="(\/[^"#?]*)"/g)) {
    const href = m[1]
    if (href.startsWith('/_next') || href.startsWith('/images') || href.startsWith('/icons')) continue
    if (/\.(css|js|svg|png|jpg|ico|txt|xml)$/.test(href)) continue
    if (!seen.has(href)) queue.push(href)
  }
}

fs.mkdirSync(OUT, { recursive: true })
let failed = 0
for (const [route, html] of results) {
  const file = path.join(OUT, routeToFile(route))
  fs.writeFileSync(file, html)
  if (MODE === 'compare') {
    const baseFile = path.join(BASELINE, routeToFile(route))
    if (!fs.existsSync(baseFile)) {
      console.error(`NEW ROUTE (no baseline): ${route}`)
      failed++
      continue
    }
    const before = fs.readFileSync(baseFile, 'utf8')
    if (before !== html) {
      console.error(`DIFF: ${route}`)
      const a = before.split('>')
      const b = html.split('>')
      for (let i = 0; i < Math.max(a.length, b.length); i++) {
        if (a[i] !== b[i]) {
          console.error(`  first divergence at chunk ${i}:`)
          console.error(`  - ${(a[i] ?? '<missing>').slice(0, 200)}`)
          console.error(`  + ${(b[i] ?? '<missing>').slice(0, 200)}`)
          break
        }
      }
      failed++
    }
  }
}
if (MODE === 'compare') {
  const baselineFiles = fs.readdirSync(BASELINE)
  for (const f of baselineFiles) {
    if (!fs.existsSync(path.join(OUT, f))) {
      console.error(`MISSING ROUTE (was in baseline): ${f}`)
      failed++
    }
  }
  console.log(failed === 0 ? `EQUIVALENT (${results.size} routes)` : `${failed} route(s) differ`)
  process.exit(failed === 0 ? 0 : 1)
}
console.log(`baseline captured: ${results.size} routes`)
```

```ts
// src/content/interpolate.ts
export type TokenSource = {
  city: string
  state: string
  phone: string
  phoneHref: string
}

export function citySlug(city: string): string {
  return city
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-|-$/g, '')
}

/**
 * Fills {tokens} in template copy. Static copy with a city mention is stored
 * ONCE in src/data with tokens; this is the only mechanism that writes a
 * city name into it — a model never edits static copy.
 */
export function t(template: string, c: TokenSource): string {
  const tokens: Record<string, string> = {
    city: c.city,
    state: c.state,
    cityLower: c.city.toLowerCase(),
    stateLower: c.state.toLowerCase(),
    citySlug: citySlug(c.city),
    phone: c.phone,
    phoneHref: c.phoneHref,
  }
  return template.replace(/\{([a-zA-Z]+)\}/g, (_, name: string) => {
    const value = tokens[name]
    if (value === undefined) throw new Error(`unknown token {${name}} in template: ${template}`)
    return value
  })
}
```

```ts
// src/content/city.ts
/*
 * The current-city accessor. Plan 1: single tenant, statically imported.
 * Plan 2 replaces the constant with per-request tenant resolution — every
 * consumer already goes through this module, so that swap is contained here.
 */
import type { CityContent } from './types'
import raw from '../../content/minneapolis.json'

export const currentCity = raw as CityContent

/** A string slot. Throws on missing/mistyped so a bad slot fails the build, not the page. */
export function s(id: string): string {
  const v = currentCity.sections[id]
  if (typeof v !== 'string') throw new Error(`content slot "${id}" missing or not a string`)
  return v
}

/** A string-list slot. */
export function sl(id: string): string[] {
  const v = currentCity.sections[id]
  if (!Array.isArray(v)) throw new Error(`content slot "${id}" missing or not a list`)
  return v
}
```
